import test from 'node:test';
import assert from 'node:assert/strict';
import { canonicalDataEqual, canonicalJson } from '../../runtime/core/data-contract.mjs';

function capture(fn) {
  try {
    return { kind: 'return', value: fn() };
  } catch (error) {
    return {
      kind: 'error',
      code: error?.code,
      message: error?.message
    };
  }
}

function oracle(left, right) {
  return capture(() => canonicalJson(left) === canonicalJson(right));
}

function compare(left, right, label) {
  const expected = oracle(left, right);
  const actual = capture(() => canonicalDataEqual(left, right));
  assert.deepStrictEqual(actual, expected, label);
}

function depthValue(depth) {
  let value = 'leaf';
  for (let i = 0; i < depth; i++) value = { next: value };
  return value;
}

test('canonicalDataEqual matches canonicalJson equality for valid data', () => {
  const shared = { id: 'shared', values: [null, false, '🌲', -0] };
  const nullPrototype = Object.create(null);
  nullPrototype.z = 3;
  nullPrototype.a = 'first';
  nullPrototype['10'] = 'ten';
  nullPrototype['2'] = 'two';

  const ownProtoKey = {};
  Object.defineProperty(ownProtoKey, '__proto__', {
    value: { retained: true }, enumerable: true, writable: true, configurable: true
  });
  const ownProtoEquivalent = {};
  Object.defineProperty(ownProtoEquivalent, '__proto__', {
    value: { retained: true }, enumerable: true, writable: true, configurable: true
  });

  const values = [
    null, false, true, '', 'plain', '雪 🌲', '\uD800', '\uDC00', '\uD800x\uDC00',
    0, -0, 1, -1, 1.25, Number.MAX_SAFE_INTEGER,
    [], [null, false, 'text', 0, -0], [[1], [2, [3]]],
    {}, { a: 1 }, { b: 2, a: 1 }, { '2': 'two', '10': 'ten', '1': 'one', z: 0 },
    nullPrototype, ownProtoKey,
    { left: shared, right: shared, repeated: [shared, shared] },
    depthValue(64)
  ];

  for (let i = 0; i < values.length; i++) {
    compare(values[i], values[i], `same valid value #${i}`);
    for (let j = 0; j < values.length; j++) compare(values[i], values[j], `valid pair ${i}/${j}`);
  }

  const insertionOrderLeft = { z: [1, { y: 'last', x: 'first' }], a: null, '12': true, '3': false };
  const insertionOrderRight = { '3': false, '12': true, a: null, z: [1, { x: 'first', y: 'last' }] };
  compare(insertionOrderLeft, insertionOrderRight, 'record insertion order and integer-looking keys');
  compare(nullPrototype, { '2': 'two', '10': 'ten', z: 3, a: 'first' }, 'null-prototype record equals ordinary record');
  compare(ownProtoKey, ownProtoEquivalent, 'own __proto__ key');
  compare({ value: -0 }, { value: 0 }, 'negative zero equals zero');
  compare(
    { a: shared, b: shared },
    { a: { id: 'shared', values: [null, false, '🌲', -0] }, b: { id: 'shared', values: [null, false, '🌲', 0] } },
    'shared references equal duplicated data'
  );
  compare({ value: { nested: [1, 2, { end: 'same' }] } }, { value: { nested: [1, 2, { end: 'same' }] } }, 'deep nested values');
});

test('canonicalDataEqual matches canonicalJson equality for differences', () => {
  const pairs = [
    [null, false, 'null differs from false'],
    [false, 0, 'boolean differs from number'],
    ['', '\u0000', 'empty string differs from nul character'],
    ['\uD800', '\uDC00', 'different isolated surrogates'],
    [[], {}, 'array differs from record'],
    [[1, 2], [1, 3], 'array tail differs'],
    [{ a: 1, z: 2 }, { a: 1, z: 3 }, 'record tail differs'],
    [{ nested: { a: [1, 2] } }, { nested: { a: [1, 2, 3] } }, 'nested tail differs'],
    [{ a: 1, b: { c: 2 } }, { a: 9, b: { c: 2 } }, 'early value differs before later equal data'],
    [{ a: 1 }, { a: 1, b: 2 }, 'record key set differs']
  ];

  for (const [left, right, label] of pairs) compare(left, right, label);
});

test('canonicalDataEqual preserves validation errors and traversal behavior', () => {
  let recordGetterCalls = 0;
  let arrayGetterCalls = 0;
  const recordGetter = { first: 0, tail: {} };
  Object.defineProperty(recordGetter.tail, 'hiddenValue', {
    enumerable: true,
    get() { recordGetterCalls++; return 1; }
  });
  const arrayGetter = [0, 1];
  Object.defineProperty(arrayGetter, '1', {
    enumerable: true,
    get() { arrayGetterCalls++; return 1; }
  });

  const sparse = [0, 1];
  delete sparse[1];
  const extraArrayProperty = [0];
  extraArrayProperty.extra = true;
  const symbolKeyRecord = { [Symbol('hidden')]: 1 };
  const nonEnumerableRecord = {};
  Object.defineProperty(nonEnumerableRecord, 'hidden', { value: 1, enumerable: false });
  const cyclic = {};
  cyclic.self = cyclic;
  class UnsupportedRecord {}

  const invalidValues = [
    NaN, Infinity, undefined, 1n, () => {}, Symbol('value'),
    recordGetter, arrayGetter, sparse, extraArrayProperty, symbolKeyRecord,
    nonEnumerableRecord, cyclic, new UnsupportedRecord(), depthValue(65)
  ];

  for (let i = 0; i < invalidValues.length; i++) {
    compare(invalidValues[i], null, `invalid left value #${i}`);
    compare(null, invalidValues[i], `invalid right value #${i}`);
  }

  assert.equal(recordGetterCalls, 0, 'record getters must never execute');
  assert.equal(arrayGetterCalls, 0, 'array getters must never execute');

  // A value mismatch cannot hide an invalid value later in the right operand.
  const rightInvalidAfterMismatch = { first: 1, tail: { deep: NaN } };
  compare({ first: 0, tail: { deep: 0 } }, rightInvalidAfterMismatch, 'right side is fully validated after an early mismatch');

  const rightGetterAfterMismatch = { first: 1, tail: {} };
  Object.defineProperty(rightGetterAfterMismatch.tail, 'hiddenValue', {
    enumerable: true,
    get() { recordGetterCalls++; return 1; }
  });
  compare({ first: 0, tail: {} }, rightGetterAfterMismatch, 'right getter after mismatch is rejected without execution');
  assert.equal(recordGetterCalls, 0, 'nested right-side getters must never execute');

  // canonicalJson(left) is evaluated first, so its error wins when both sides are invalid.
  const leftInvalid = { first: NaN };
  const rightInvalid = { first: undefined };
  compare(leftInvalid, rightInvalid, 'left validation error has precedence');
});
